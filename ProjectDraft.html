import requests
import pandas as pd
from bs4 import BeautifulSoup
from io import StringIO
import re
import time
from IPython.display import display

def clean_season_table(raw_df, year):
    df = raw_df.copy()
    for col in df.columns:
        df[col] = df[col].astype(str).str.replace(
            rf'^{re.escape(col)}\s*', '', regex=True, flags=re.IGNORECASE
        ).str.strip()
    df['Year'] = year
    return df

all_champions = []
years_with_no_data = []
years_failed = []

start_year = 1928
end_year = 2026

for year in range(start_year, end_year + 1):
    url = f"https://api.nwhof.org/national-wrestling-hall-of-fame/champions-database?school=&season={year}&wrestler="

    try:
        response = requests.get(url, timeout=10)
        tables = pd.read_html(StringIO(response.text))

        if len(tables) > 0 and len(tables[0]) > 0:
            df = clean_season_table(tables[0], year)
            all_champions.append(df)

            df.to_csv(f'champions_{year}.csv', index=False)  # one CSV per year
            print(f"{year}: got {len(df)} rows — saved champions_{year}.csv")
        else:
            years_with_no_data.append(year)
            print(f"{year}: no data")

    except Exception as e:
        years_failed.append(year)
        print(f"{year}: FAILED — {e}")

    time.sleep(0.5)  # be polite to their server

# combine every year into one table
champions_df = pd.concat(all_champions, ignore_index=True)

print(f"\nTotal rows: {len(champions_df)}")
print(f"Years with no data: {years_with_no_data}")
print(f"Years that failed: {years_failed}")

plt.title(f"Championship Concentration: {n_schools_80pct} of {len(school_counts)} "

plt.tight_layout()
plt.show()

import matplotlib.pyplot as plt

school_counts = combined["School"].value_counts()

TOP_N = 25  # adjust to taste - more schools = more crowded x-axis

top_schools = school_counts.head(TOP_N)
others_total = school_counts[TOP_N:].sum()

# what the chart will show: top N named schools + one "All others" bar
plot_counts = pd.concat([top_schools, pd.Series({"All other schools": others_total})])

# cumulative % should still be based on the FULL ranked list, not just the top N,
# so the 80% line stays accurate
full_cum_pct = school_counts.cumsum() / school_counts.sum() * 100
cum_pct_for_plot = list(full_cum_pct.head(TOP_N).values) + [100.0]

fig, ax1 = plt.subplots(figsize=(14, 7))

x = range(len(plot_counts))
ax1.bar(x, plot_counts.values, color="steelblue")
ax1.set_ylabel("Number of Championships")
ax1.set_xticks(x)
ax1.set_xticklabels(plot_counts.index, rotation=60, ha="right")

ax2 = ax1.twinx()
ax2.plot(x, cum_pct_for_plot, color="firebrick", linewidth=2, marker="o", markersize=4)
ax2.set_ylabel("Cumulative % of ALL championships (all schools)")
ax2.axhline(80, color="gray", linestyle="--", linewidth=1)

n_schools_80pct = (full_cum_pct <= 80).sum()
plt.title(f"Championship Concentration: Top {TOP_N} Schools "
          f"({n_schools_80pct} schools = 80% of all titles)")
plt.tight_layout()
